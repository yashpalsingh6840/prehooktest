using Etl.Core.Abstractions;
using LoadReferenceData.Csv;
using LoadReferenceData.Model;
using LoadReferenceData.Ssis;

namespace LoadReferenceData.Mapping;

public sealed class DepartmentTransform : IRowTransform<DepartmentCsvRow, Department>
{
    public Department Map(DepartmentCsvRow row, in RowContext ctx) => new()
    {
        DepartmentID = row.DepartmentID,
        DepartmentCode = row.DepartmentCode,
        DepartmentName = row.DepartmentName,
        CostCenter = row.CostCenter,
        HeadCount = row.HeadCount,
        DepartmentKey = DER_DepartmentKey.DepartmentKey(row, ctx),
        LoadedAtUtc = DER_DepartmentKey.LoadedAtUtc(row, ctx),
    };
}
