using Etl.Core.Abstractions;
using Etl.Core.Ssis;
using LoadReferenceData.Csv;
using LoadReferenceData.Model;
using LoadReferenceData.Ssis;

namespace LoadReferenceData.Mapping;

internal static class DER_DepartmentKey
{
    // (DT_WSTR,25)(UPPER(DepartmentCode) + "-" + (DT_WSTR,10)DepartmentID)
    public static string DepartmentKey(DepartmentCsvRow row, in RowContext ctx) => WidthGuard.Wstr($"{SsisFn.Upper(row.DepartmentCode)}-{SsisFn.Str(row.DepartmentID)}", 25, nameof(Department.DepartmentKey), ctx.RowNumber);

    // GETUTCDATE()
    public static DateTime LoadedAtUtc(DepartmentCsvRow row, in RowContext ctx) => ctx.LoadedAtUtc;
}
