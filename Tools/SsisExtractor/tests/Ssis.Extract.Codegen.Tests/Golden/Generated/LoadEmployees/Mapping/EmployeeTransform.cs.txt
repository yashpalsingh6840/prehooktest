using Etl.Core.Abstractions;
using LoadEmployees.Csv;
using LoadEmployees.Model;
using LoadEmployees.Ssis;

namespace LoadEmployees.Mapping;

public sealed class EmployeeTransform : IRowTransform<EmployeeCsvRow, Employee>
{
    public Employee Map(EmployeeCsvRow row, in RowContext ctx) => new()
    {
        EmployeeID = row.EmployeeID,
        Department = row.Department,
        HireDate = row.HireDate,
        Salary = row.Salary,
        FullName = DER_MergeColumns.FullName(row, ctx),
        Location = DER_MergeColumns.Location(row, ctx),
        EmployeeKey = DER_MergeColumns.EmployeeKey(row, ctx),
        LoadedAtUtc = DER_MergeColumns.LoadedAtUtc(row, ctx),
    };
}
