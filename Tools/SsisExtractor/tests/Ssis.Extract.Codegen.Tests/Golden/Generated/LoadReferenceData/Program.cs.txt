using Etl.Core.Hosting;
using LoadReferenceData;
using LoadReferenceData.Model;
using Microsoft.Extensions.DependencyInjection;

const string PackageName = "LoadReferenceData";

var builder = EtlHost.Create(args, PackageName);
builder.Services.AddEtlDbContext<LoadReferenceDataDbContext>();

using var host = builder.Build();
return await new LoadReferenceDataPackage(host.Services).RunAsync(CancellationToken.None);
